import math

# CLASS: ParkedCar
# PURPOSE: Store and provide car information

class ParkedCar:
    def __init__(self, make, model, color, license_number, minutes_parked):
        
        # Initialize car details with required attributes
        self.__make = make
        self.__model = model
        self.__color = color
        self.__license_number = license_number
        self.__minutes_parked = minutes_parked
    
    def get_make(self):
        # RETURN car make
        return self.__make
    
    def get_model(self):
        # RETURN car model
        return self.__model
    
    def get_color(self):
        # RETURN car color
        return self.__color
    
    def get_license_number(self):
        # RETURN license number
        return self.__license_number
    
    def get_minutes_parked(self):
        # RETURN minutes parked
        return self.__minutes_parked

    
# CLASS: ParkingMeter
# PURPOSE: Store purchased parking time of meter

class ParkingMeter:
    def __init__(self, minutes_purchased):
        
        # Store purchased parking time in minutes
        self.__minutes_purchased = minutes_purchased
    
    def get_minutes_purchased(self):
        # RETURN purchased minutes
        return self.__minutes_purchased


# CLASS: ParkingTicket
# PURPOSE: Represent and display a violation

class ParkingTicket:
    def __init__(self, car, minutes_over, officer_name, officer_badge):
        
        # Store ticket details and calculate fine
        self.__car = car
        self.__minutes_over = minutes_over
        self.__officer_name = officer_name
        self.__officer_badge = officer_badge
        self.__fine = self.calculate_fine()
    
    def calculate_fine(self):
        
        # hours_over = CEILING(minutes_over / 60)
        # IF hours_over <= 1 → fine = 25
        # ELSE → fine = 25 + (hours_over - 1) * 10
        hours_over = math.ceil(self.__minutes_over / 60)
        
        if hours_over <= 1:
            return 25
        else:
            return 25 + (hours_over - 1) * 10
    
    def display_ticket(self):
        
        # PRINT all ticket details including car info and fine
        print("PARKING TICKET")
        print(f"Make: {self.__car.get_make()}")
        print(f"Model: {self.__car.get_model()}")
        print(f"Color: {self.__car.get_color()}")
        print(f"License Number: {self.__car.get_license_number()}")
        print(f"Minutes Parked: {self.__car.get_minutes_parked()}")
        print(f"Minutes Over Limit: {self.__minutes_over}")
        print(f"Fine Amount: ${self.__fine}")
        print(f"Officer Name: {self.__officer_name}")
        print(f"Officer Badge: {self.__officer_badge}")
        print("-" * 50)

class PoliceOfficer:
    """Represents a police officer who inspects parked cars for violations."""
    def __init__(self, name, badge_number):
        """
        Initialize a police officer.
        
        Args:
            name: Officer's name
            badge_number: Officer's badge number
        """
        self.name = name
        self.badge_number = badge_number
    
    def get_name(self):
        return self.name
    
    def get_badge_number(self):
        return self.badge_number
    
    def run_inspection(self, car, meter):
        """
        Inspect a parked car and its meter for violations.
        
        Args:
            car: ParkedCar object to inspect
            meter: ParkingMeter object associated with the car
        
        Returns:
            ParkingTicket object if violation found, None otherwise
        """
        minutes_parked = car.get_minutes_parked()
        minutes_purchased = meter.get_minutes_purchased()
        
        if minutes_parked > minutes_purchased:
            minutes_over = minutes_parked - minutes_purchased
            return ParkingTicket(car, minutes_over, self.name, self.badge_number)
        
        return None
